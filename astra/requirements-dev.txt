# requirements-dev.txt — test/development dependencies.
# =============================================================================
# requirements.txt is the RUNTIME manifest: it is what a user (and the
# PyInstaller sidecar) installs, and it deliberately carries no test tooling.
# CI and contributors need this file as well:
#
#   pip install -r requirements.txt -r requirements-dev.txt
#   python3 -m pytest tests/ -q
#
# Without it `python3 -m pytest` fails with "No module named pytest" and the
# suite never runs — which is exactly how the backend CI job was failing.
# =============================================================================

-r requirements.txt

                           #   need pandas — that is the point.
