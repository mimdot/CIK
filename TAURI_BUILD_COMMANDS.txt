# Tauri Build Commands for Career Intelligence Kit
# ================================================
# Run these commands in your terminal

## 1. Install System Dependencies (Ubuntu/Debian)
# Copy and paste this entire block into your terminal:

sudo apt-get update && sudo apt-get install -y \
  libwebkit2gtk-4.1-dev \
  libappindicator3-dev \
  librsvg2-dev \
  patchelf \
  libgtk-3-dev \
  libglib2.0-dev \
  libsoup2.4-dev \
  libjavascriptcoregtk-4.1-dev \
  pkg-config \
  python3-pip

## 2. Build the Tauri App
cd /home/mohammad-reza/career_intelligence_kit/dashboard
npm run tauri build

## 3. Find the built AppImage
# After successful build, the AppImage will be at:
# /home/mohammad-reza/career_intelligence_kit/dashboard/src-tauri/target/release/bundle/appimage/career-intelligence_0.1.0_amd64.AppImage

## 4. Alternative: Trigger GitHub Actions Build (no local Rust needed)
# Push a tag to trigger automated multi-platform build:

git add .github/workflows/tauri-release.yml
git commit -m "ci: add Tauri system dependencies for Linux/macOS/Windows"
git tag v0.1.0
git push origin v0.1.0

# This creates a GitHub Release with:
# - Linux: .AppImage
# - macOS: .dmg (Intel + ARM)
# - Windows: .msi + .exe

## 5. If you get "cargo not found" error:
# Install Rust first:
curl --proto '=https' --tlsv1.2 -sSf https://sh.rustup.rs | sh
source ~/.cargo/env
cargo --version
